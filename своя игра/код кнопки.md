#айди {
  /* 1. ДЕЛАЕМ КНОПКУ КРУГЛОЙ (Ширина и высота строго одинаковые) */
  width: 120px;
  height: 120px;
  border-radius: 50%;

  /* 2. Включаем режим перемещения */

  /* 3. УПРАВЛЕНИЕ МЕСТОМ (крути эти цифры, чтобы двигать кнопку) */
  top: -20px;  /* Минус двигает ВВЕРХ, плюс — ВНИЗ */
  left: 50px;  /* Плюс двигает ВПРАВО, минус — ВЛЕВО */

  /* 4. КРАСИВЫЙ ВИД КНОПКИ */
  background-color: #ff4757; /* Яркий цвет фона */
  color: white;              /* Белый цвет текста */

  font-weight: bold;         /* Жирный шрифт текста */
  border: none;              /* Убираем некрасивую стандартную рамку */
  cursor: pointer;           /* Появляется палец при наведении мышки */
}
